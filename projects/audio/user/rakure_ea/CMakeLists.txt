# MSVC currently gates its C11 atomic implementation behind this C-only option.
# The portable chain/GUI do not require atomics or this compiler option.
function(rea_require_c11_atomics target)
    target_compile_features(${target} PRIVATE c_std_11)
    set_target_properties(${target} PROPERTIES C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:C>:/experimental:c11atomics>)
    endif()
endfunction()
