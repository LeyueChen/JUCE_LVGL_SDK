# Portable core; only the optional SDK port depends on FreeRTOS/SDK services.
set(rea_neural_default OFF)
if(TARGET sdk_rtneural)
    set(rea_neural_default ON)
endif()
option(REA_ENABLE_RTNEURAL "Enable the RTNeural Dense demonstration" ${rea_neural_default})
if(DEFINED CONFIG_SDK_RTNEURAL AND NOT CONFIG_SDK_RTNEURAL STREQUAL "y")
    # The SDK's component switch remains authoritative for its provider.
    set(REA_ENABLE_RTNEURAL OFF)
endif()
set(REA_RTNEURAL_TARGET "sdk_rtneural" CACHE STRING "Target providing RTNeural")
add_library(rakure_ea STATIC src/rea_chain.cpp src/rea_neural.cpp)
add_library(rakure::ea ALIAS rakure_ea)
target_compile_features(rakure_ea PUBLIC cxx_std_17)
target_include_directories(rakure_ea PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(rakure_ea PUBLIC REA_USE_RTNEURAL=$<BOOL:${REA_ENABLE_RTNEURAL}>)
if(REA_ENABLE_RTNEURAL)
    if(NOT TARGET ${REA_RTNEURAL_TARGET})
        message(FATAL_ERROR "Provide REA_RTNEURAL_TARGET or disable REA_ENABLE_RTNEURAL")
    endif()
    target_link_libraries(rakure_ea PRIVATE ${REA_RTNEURAL_TARGET})
endif()
set(rea_sdk_default OFF)
if(TARGET sdk::runtime)
    set(rea_sdk_default ON)
endif()
option(REA_BUILD_SDK_PORT "Build the SDK/FreeRTOS audio task and controls" ${rea_sdk_default})
set(module_targets rakure::ea)
if(REA_BUILD_SDK_PORT)
    if(NOT TARGET sdk::runtime)
        message(FATAL_ERROR "REA_BUILD_SDK_PORT requires sdk::runtime")
    endif()
    add_library(rakure_ea_sdk STATIC ports/sdk/rea_audio_task.cpp ports/sdk/rea_control.c)
    add_library(rakure::ea_sdk ALIAS rakure_ea_sdk)
    target_link_libraries(rakure_ea_sdk PUBLIC rakure::ea sdk::runtime)
    list(APPEND module_targets rakure::ea_sdk)
endif()
set_property(DIRECTORY PROPERTY SDK_USER_MODULE_TARGET "${module_targets}")
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
