set(RAKURE_GUI_LVGL_TARGET "lvgl" CACHE STRING "Target providing LVGL")
if(NOT TARGET ${RAKURE_GUI_LVGL_TARGET})
    message(FATAL_ERROR "Provide RAKURE_GUI_LVGL_TARGET before adding rakure_gui")
endif()
add_library(rakure_gui STATIC src/rakure_gui.c)
add_library(rakure::gui ALIAS rakure_gui)
target_include_directories(rakure_gui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(rakure_gui PRIVATE c_std_11)
target_link_libraries(rakure_gui PUBLIC ${RAKURE_GUI_LVGL_TARGET})
set_property(DIRECTORY PROPERTY SDK_USER_MODULE_TARGET rakure::gui)
