cmake_minimum_required(VERSION 3.25)
# Embed compiler debug information in object files; avoid the shared PDB compiler service.
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>" CACHE STRING "MSVC debug format")
project(LVGL_JUCE_SDK VERSION 0.2.0 LANGUAGES C CXX)
option(BUILD_TESTING "Build SDK verification programs and host diagnostics" OFF)
option(SDK_ENABLE_WINDOW_TESTS "Register opt-in JUCE window/device/viewport tests" OFF)
if(MSVC AND DEFINED SDK_MSVC_INCLUDE_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SDK_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SDK_PROJECT_DIR "${CMAKE_SOURCE_DIR}/examples/basic" CACHE PATH "Selected project directory")
file(READ "${SDK_PROJECT_DIR}/project.json" SDK_PROJECT_METADATA)
string(JSON SDK_PROJECT_ID GET "${SDK_PROJECT_METADATA}" id)
string(JSON SDK_USER_RELATIVE GET "${SDK_PROJECT_METADATA}" user_dir)
set(SDK_APP_DIR "${SDK_PROJECT_DIR}/${SDK_USER_RELATIVE}")
set(SDK_BOARD windows_sim CACHE STRING "Board")
set(SDK_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated" CACHE PATH "Generated configuration")
find_package(Python COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/configure.py"
    --project-dir "${SDK_PROJECT_DIR}" --board "${SDK_BOARD}"
    --output "${SDK_GENERATED_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/Kconfig" "${CMAKE_SOURCE_DIR}/boards/windows_sim/defconfig"
    "${SDK_PROJECT_DIR}/project.json" "${SDK_APP_DIR}/app.conf"
    "${CMAKE_SOURCE_DIR}/third_party/components.json" "${CMAKE_SOURCE_DIR}/.gitmodules")
if(EXISTS "${SDK_PROJECT_DIR}/targets/${SDK_BOARD}.conf")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SDK_PROJECT_DIR}/targets/${SDK_BOARD}.conf")
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/.local/config/${SDK_PROJECT_ID}/${SDK_BOARD}.conf")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/.local/config/${SDK_PROJECT_ID}/${SDK_BOARD}.conf")
endif()
set(SDK_DATA_DIRECTORY "${CMAKE_SOURCE_DIR}/.local/data/${SDK_PROJECT_ID}/${SDK_BOARD}")
configure_file(sdk/config/project.h.in "${SDK_GENERATED_DIR}/sdk_project.h" @ONLY)
include("${SDK_GENERATED_DIR}/config.cmake")
if(NOT WIN32 OR NOT SDK_BOARD STREQUAL "windows_sim")
    message(FATAL_ERROR "Only windows_sim is implemented. Embedded providers require a board/BSP integration.")
endif()
add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
    add_compile_options(/utf-8 /Zc:preprocessor)
endif()
include(cmake/Dependencies.cmake)
include(cmake/Components.cmake)
include(third_party/recipes/esp-reference.cmake)
include(cmake/Protocol.cmake)
add_library(sdk_storage_contract STATIC sdk/services/storage/contract.c)
target_include_directories(sdk_storage_contract PUBLIC sdk/include)
target_link_libraries(sdk_components INTERFACE sdk_storage_contract)

add_library(sdk_media STATIC sdk/services/media/media.cpp)
target_include_directories(sdk_media PUBLIC sdk/include PRIVATE sdk/services/media "${SDK_GENERATED_DIR}")

add_library(sdk_runtime STATIC
    sdk/runtime/core/core.cpp
    sdk/osal/freertos/os.c
    sdk/services/runtime.c
    sdk/services/input/buttons.c
    platforms/windows/devices/input.cpp
    sdk/adapters/lvgl/display.c
    platforms/windows/devices/storage_backend.cpp
    platforms/windows/core/host_bridge.cpp
    platforms/windows/freertos/runtime.cpp)
target_include_directories(sdk_runtime PUBLIC sdk/include "${SDK_GENERATED_DIR}" PRIVATE platforms/windows/devices)
target_link_libraries(sdk_runtime PUBLIC sdk_freertos lvgl sdk_components sdk_media)
add_library(sdk::runtime ALIAS sdk_runtime)

if(BUILD_TESTING)
    enable_testing()
    add_library(sdk_rtos_test_support STATIC tests/rtos.c)
    target_link_libraries(sdk_rtos_test_support PRIVATE sdk_runtime)
    target_compile_definitions(sdk_runtime PRIVATE SDK_DIAGNOSTICS=1)
endif()
add_subdirectory("${SDK_APP_DIR}" "${CMAKE_BINARY_DIR}/user")
add_subdirectory("${SDK_JUCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
juce_add_gui_app(lvgl_juce PRODUCT_NAME "LVGL JUCE SDK" COMPANY_NAME "LVGL JUCE SDK" BUNDLE_ID "dev.lvgljuce.simulator")
target_sources(lvgl_juce PRIVATE platforms/windows/host_juce/Main.cpp platforms/windows/host_juce/DisplayCanvas.cpp platforms/windows/audio/HostMedia.cpp)
target_include_directories(lvgl_juce PRIVATE platforms/windows/audio platforms/windows/host_juce sdk/services/media)
target_compile_definitions(lvgl_juce PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0)
target_link_libraries(lvgl_juce PRIVATE sdk_runtime sdk_user juce::juce_gui_basics juce::juce_audio_utils juce::juce_recommended_config_flags)
set_target_properties(lvgl_juce PROPERTIES OUTPUT_NAME lvgl_juce RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

if(BUILD_TESTING)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/disabled")
file(WRITE "${CMAKE_BINARY_DIR}/disabled/sdk_config.h" "#pragma once\n")
add_executable(sdk_disabled_tests tests/disabled.cpp sdk/services/media/media.cpp)
target_include_directories(sdk_disabled_tests PRIVATE "${CMAKE_BINARY_DIR}/disabled" sdk/include sdk/services/media)
add_test(NAME disabled_media COMMAND sdk_disabled_tests)
target_sources(lvgl_juce PRIVATE tests/viewport.cpp)
target_link_libraries(lvgl_juce PRIVATE sdk_rtos_test_support)

target_include_directories(lvgl_juce PRIVATE tests)
target_compile_definitions(lvgl_juce PRIVATE SDK_DIAGNOSTICS=1)
add_executable(sdk_component_tests tests/components.cpp platforms/windows/devices/storage_backend.cpp)
target_include_directories(sdk_component_tests PRIVATE sdk/include platforms/windows/devices "${SDK_GENERATED_DIR}")
target_link_libraries(sdk_component_tests PRIVATE sdk_components)
if(CONFIG_SDK_NANOPB STREQUAL "y")
    sdk_generate_protocol(sdk_test_protocol "${CMAKE_SOURCE_DIR}/tests/protocol" "${SDK_GENERATED_DIR}/test-protocol")
    target_link_libraries(sdk_component_tests PRIVATE sdk_test_protocol)
endif()
add_test(NAME portable_components COMMAND sdk_component_tests "${CMAKE_BINARY_DIR}/test-data")
set_tests_properties(portable_components PROPERTIES TIMEOUT 30)
add_test(NAME configuration_contracts COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_config.py")
add_test(NAME project_isolation COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_projects.py")
add_test(NAME copied_user_modules COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_modules.py" --generated "${SDK_GENERATED_DIR}")
set_tests_properties(copied_user_modules PROPERTIES TIMEOUT 90)
add_test(NAME repository_safety COMMAND "${Python_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_repository.py")
add_executable(sdk_media_tests tests/media.cpp tests/media_c_processor.c)
target_include_directories(sdk_media_tests PRIVATE sdk/services/media "${SDK_GENERATED_DIR}")
target_link_libraries(sdk_media_tests PRIVATE sdk_media)
add_test(NAME audio_midi_contracts COMMAND sdk_media_tests)
set_tests_properties(audio_midi_contracts PROPERTIES TIMEOUT 30)
if(SDK_ENABLE_WINDOW_TESTS)
add_test(NAME host_device_settings COMMAND lvgl_juce --media-self-test "${CMAKE_BINARY_DIR}/device-tests")
set_tests_properties(host_device_settings PROPERTIES TIMEOUT 45)
add_test(NAME runtime_smoke COMMAND lvgl_juce --smoke-seconds 4 --no-audio --data-dir "${CMAKE_BINARY_DIR}/runtime-test-data")
set_tests_properties(runtime_smoke PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}")
add_test(NAME viewport_contracts COMMAND lvgl_juce --viewport-self-test "${CMAKE_BINARY_DIR}/viewport-tests")
set_tests_properties(viewport_contracts PROPERTIES TIMEOUT 30)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/viewport-runtime")
add_test(NAME viewport_runtime COMMAND lvgl_juce --viewport-smoke --smoke-seconds 7 --no-audio --data-dir "${CMAKE_BINARY_DIR}/viewport-runtime/data")
set_tests_properties(viewport_runtime PROPERTIES TIMEOUT 35 WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/viewport-runtime")
set_tests_properties(host_device_settings runtime_smoke viewport_contracts viewport_runtime PROPERTIES LABELS "window")
endif()
endif()
